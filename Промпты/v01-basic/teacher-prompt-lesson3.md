3. **Строительные блоки** (новый раздел):
```python
import numpy as np
import matplotlib.pyplot as plt

# Создаем класс для нашего нейрона
class SimpleNeuron:
    def __init__(self, weights, bias):
        self.weights = weights
        self.bias = bias
    
    def activate(self, x):
        return 1 / (1 + np.exp(-x))  # Sigmoid
    
    def predict(self, inputs):
        sum_weighted = np.dot(inputs, self.weights) + self.bias
        return self.activate(sum_weighted)

# Создаем нейрон для принятия решения о пляже
beach_neuron = SimpleNeuron(
    weights=[0.6, -0.7, 0.3],  # Важность: температура, дождь, выходной
    bias=-0.1
)

# Тестируем разные ситуации
situations = [
    [0.9, 0.1, 1.0],  # Жарко, нет дождя, выходной
    [0.2, 0.8, 1.0],  # Холодно, дождь, выходной
    [0.7, 0.3, 0.0],  # Тепло, небольшой дождь, рабочий день
]

for s in situations:
    decision = beach_neuron.predict(s)
    print(f"Ситуация: {s}")
    print(f"Решение идти на пляж: {decision:.2%}\n")

"""
🔬 Эксперименты:
1. Измени веса - как изменится решение?
2. Добавь новый фактор (например, наличие свободного времени)
3. Попробуй разные ситуации

Какие выводы ты можешь сделать о работе нейрона?
```
