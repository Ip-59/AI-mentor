# Создаем простые данные
# Представим, что это информация о погоде и решениях о прогулке

    [25, 0],  # [температура, дождь]
    [30, 0],
    [20, 1],
    [15, 1]
])
decisions = ['гулять', 'гулять', 'дома', 'дома']

# Создаем и обучаем модель

model.fit(data, decisions)

# Пробуем предсказать решение для новой погоды

print(f"Я думаю, надо {prediction[0]}!")

8. **Простой проект: AI-погодный советник**:
```python
import pandas as pd
from sklearn.tree import DecisionTreeClassifier
import numpy as np

# Создаем более сложный датасет о погоде и активностях
weather_data = pd.DataFrame({
    'температура': [25, 30, 20, 15, 28, 22, 18],
    'дождь': [0, 0, 1, 1, 0, 0, 1],
    'ветер': [10, 15, 20, 25, 12, 18, 22],
    'активность': ['парк', 'пляж', 'дома', 'дома', 'парк', 'спорт', 'дома']
})

# Подготовим данные для модели
X = weather_data[['температура', 'дождь', 'ветер']]
y = weather_data['активность']

# Создаем и обучаем модель
model = DecisionTreeClassifier()
model.fit(X, y)

# Функция для получения рекомендации
def get_activity_recommendation(temp, rain, wind):
    prediction = model.predict([[temp, rain, wind]])
    return prediction[0]

# Интерактивное тестирование
test_conditions = [
    [24, 0, 12],  # Тепло, без дождя, легкий ветер
    [31, 0, 8],   # Жарко, без дождя, слабый ветер
    [19, 1, 25]   # Прохладно, дождь, сильный ветер
]

for condition in test_conditions:
    activity = get_activity_recommendation(*condition)
    print(f"При температуре {condition[0]}°C, "
          f"{'дождливой' if condition[1] else 'ясной'} погоде, "
          f"и ветре {condition[2]} м/с "
          f"рекомендую: {activity}")

"""
🎯 Попробуй:
1. Добавить новые активности
2. Изменить условия погоды
3. Добавить новые параметры (например, время суток)

Как ты думаешь, почему модель дает именно такие рекомендации?
"""
```
