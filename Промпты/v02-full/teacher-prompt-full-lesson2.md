```python
# Давай создадим простой пример ML с использованием знакомого Python
import numpy as np

# Наши обучающие данные
X = np.array([2, 3, 4, 5])  # входные числа
y = np.array([4, 9, 16, 25])  # выходные числа

# Простая модель: y = ax^2
def simple_model(x, a):
    return a * x**2

# Проверка ошибки модели
def check_error(a):
    predictions = simple_model(X, a)
    error = np.mean((predictions - y)**2)
    return error

# Попробуем разные значения параметра a
for a in [0.5, 1.0, 1.5, 2.0]:
    error = check_error(a)
    print(f"При a={a}, ошибка = {error:.2f}")

"""
🤔 Подумай:
1. Какое значение 'a' дает наименьшую ошибку?
2. Почему мы используем квадрат разности для ошибки?
3. Как это похоже на то, как ты сам искал закономерность?
"""
```

6. **Практическое задание: Анализ данных о погоде**:
```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

# Создаем датасет о погоде
np.random.seed(42)
n_samples = 100

data = pd.DataFrame({
    'температура': np.random.uniform(15, 30, n_samples),
    'влажность': np.random.uniform(30, 90, n_samples),
    'давление': np.random.uniform(740, 760, n_samples)
})

# Создаем целевую переменную (вероятность дождя)
data['вероятность_дождя'] = (0.3 * data['влажность'] +
                            -0.2 * data['температура'] +
                            0.1 * data['давление'] +
                            np.random.normal(0, 5, n_samples))
data['вероятность_дождя'] = data['вероятность_дождя'].clip(0, 100)

# Разделяем данные
X = data[['температура', 'влажность', 'давление']]
y = data['вероятность_дождя']
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

# Обучаем модель
model = LinearRegression()
model.fit(X_train, y_train)

# Делаем предсказания
predictions = model.predict(X_test)

# Визуализируем результаты
plt.figure(figsize=(10, 6))
plt.scatter(y_test, predictions, alpha=0.5)
plt.plot([0, 100], [0, 100], 'r--')
plt.xlabel('Реальные значения')
plt.ylabel('Предсказания')
plt.title('Сравнение предсказаний с реальными данными')
plt.show()
```
