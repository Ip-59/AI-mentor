4. **Практическая часть с кодом**:
```python
import numpy as np

class BeachNeuron:
    def __init__(self, weights, bias):
        self.weights = weights
        self.bias = bias

    def activate(self, x):
        return 1 / (1 + np.exp(-x))

    def decide(self, temperature, rain, is_weekend):
        inputs = np.array([temperature, rain, is_weekend])
        sum_weighted = np.dot(inputs, self.weights) + self.bias
        return self.activate(sum_weighted)

# Создаем нейрон
my_neuron = BeachNeuron(
    weights=[0.6, -0.7, 0.3],  # Важность факторов
    bias=-0.1
)

# Пробуем разные ситуации
situations = [
    [0.9, 0.1, 1],  # Жарко, ясно, выходной
    [0.2, 0.8, 1],  # Холодно, дождь, выходной
    [0.7, 0.3, 0],  # Тепло, небольшой дождь, будний день
]

for temp, rain, weekend in situations:
    decision = my_neuron.decide(temp, rain, weekend)
    print(f"Температура: {temp*100}°C, Дождь: {rain*100}%, {'Выходной' if weekend else 'Будний'}")
    print(f"Вероятность похода на пляж: {decision:.1%}\n")
```

6. **Практический пример с распознаванием цифр**:
```python
from sklearn.datasets import load_digits
from sklearn.neural_network import MLPClassifier
from sklearn.model_selection import train_test_split
import matplotlib.pyplot as plt

# Загружаем данные
digits = load_digits()
X_train, X_test, y_train, y_test = train_test_split(digits.data, digits.target)

# Создаем простую сеть
model = MLPClassifier(
    hidden_layer_sizes=(32, 16),
    activation='relu',
    max_iter=1000
)

# Обучаем
model.fit(X_train, y_train)

# Проверяем точность
score = model.score(X_test, y_test)
print(f"Точность модели: {score:.2%}")

# Визуализируем результаты
def show_predictions(n_samples=5):
    fig, axes = plt.subplots(2, n_samples, figsize=(12, 4))
    for i in range(n_samples):
        axes[0, i].imshow(X_test[i].reshape(8, 8), cmap='gray')
        axes[0, i].axis('off')
        pred = model.predict([X_test[i]])[0]
        axes[1, i].text(0.5, 0.5, f'Это цифра: {pred}', ha='center')
        axes[1, i].axis('off')
    plt.show()

show_predictions()
```
